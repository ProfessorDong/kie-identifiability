% so no rescaling occurs in the manuscript.  Until 2026-10-20 this was one panel

  asym/.style={icig, scale only axis, width=31mm,
    xmin=-0.50, xmax=0.20, ytick=\empty, ymajorgrids=false,

    xtick={-0.4,-0.2,0,0.2}, minor x tick num=1,
    xticklabel style={/pgf/number format/.cd, fixed, precision=1, /tikz/.cd},

}
\def\faPanel#1#2#3{%
  % #1 rows, #2 data file, #3 panel letter
  \addplot[draw=none, fill=cGrey!10] coordinates
    {(-0.50,0.3) (-0.042086,0.3) (-0.042086,#1+0.7) (-0.50,#1+0.7)} \closedcycle;
  \addplot[draw=none, fill=cBandG!13] coordinates
    {(-0.042086,0.3) (0.20,0.3) (0.20,#1+0.7) (-0.042086,#1+0.7)} \closedcycle;
  \input{#2}
  \draw[cBandG!60!black, line width=0.6pt] (axis cs:-0.042086,0.3) --
       (axis cs:-0.042086,#1+0.7);
  \draw[cInk, line width=0.5pt, dashed] (axis cs:0,0.3) -- (axis cs:0,#1+0.7);
}

\begin{axis}[asym, name=A, height=\faRowsA*2.95mm, ymin=0.3, ymax=\faRowsA+0.7]
